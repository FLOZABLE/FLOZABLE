let maxTags = 10,
  tag = [];

countTags();
createTag();

function countTags() {
  input.focus();
  tagNumb.innerText = maxTags - tag.length;
}

function createTag() {
  ul.querySelectorAll("li").forEach((li) => li.remove());
  tag
    .slice()
    .reverse()
    .forEach((tag) => {
      let liTag = `<li><p class = "tag">${tag}</p> <i class="uit uit-multiply" onclick="remove(this, '${tag}')"></i></li>`;
      ul.insertAdjacentHTML("afterbegin", liTag);
    });
  countTags();
}

function remove(element, tag) {
  let index = tag.indexOf(tag);
  tag = [...tag.slice(0, index), ...tag.slice(index + 1)];
  element.parentElement.remove();
  countTags();
}

function addTag(e) {
  if (e.key == "Enter") {
    let tag = e.target.value.replace(/\s+/g, " ");
    if (tag.length > 1 && !tag.includes(tag)) {
      if (tag.length < 10) {
        tag.split(",").forEach((tag) => {
          tag.push(tag);
          createTag();
        });
      }
    }
    e.target.value = "";
  }
}

input.addEventListener("keyup", addTag);

const removeBtn = document.querySelector(".details button");
removeBtn.addEventListener("click", () => {
  tag.length = 0;
  ul.querySelectorAll("li").forEach((li) => li.remove());
  countTags();
});
